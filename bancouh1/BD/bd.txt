-- creacion de de la baes de datos BANCOUH
create database BancoUH

use BancoUH

-- creacion de las tabla

create table Usuarios
(
   correo varchar(50) primary key,
   clave varchar(50) not null,
   nombre varchar(100)
)


-- Insertar usuarios
Insert into USuarios (correo, clave, nombre) values (@correo,@clave,@nombre'),
                                                    ('Jose@uh.com','456','Jose Ramirez')
Insert into USuarios (correo, clave, nombre) values ('Suana@uh.com','487','Susana Jimenez')

-- consultar usuarios

Select * from usuarios
Select correo, clave, nombre from usuarios
select correo from usuarios
select * from usuarios where correo = 'maria@uh.com' -- filtro se usa el la palabra "where"

-- Actualizar usuarios

update Usuarios set correo = 'Karla@uh.com', clave='999' where correo = 'Karka@uh.com'

-- borrar usuarios

delete usuarios where correo = 'jose@uh.com'



create table Tecnicos
(
   tecnicoID varchar(50) primary key not null,
   nombre varchar(100) not null,
   especialidad varchar(50) not null,
)





create table Equipos
(
   
   EquipoID varchar(50) primary key not null,
   TipoEquipo varchar(100)not null,
   Modelo varchar(100) not null,
   correo varchar(50) not null,

   FOREIGN key(correo) REFERENCES Usuarios(correo)

)
